set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test)
function(add_lingmo_session_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE Lingmo::Session Qt6::Test)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
add_lingmo_session_test(tst_SessionManager)
add_lingmo_session_test(tst_CrashRecovery)
add_lingmo_session_test(tst_Component)
